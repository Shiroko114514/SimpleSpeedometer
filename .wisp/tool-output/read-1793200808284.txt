  98|   /// 当定位流超过 3 秒没有产生新数据时，主动拉取一次当前位置，

 100|   void _startWatchdog() {
 101|     _watchdog?.cancel();
 102|     _watchdog = Timer.periodic(const Duration(seconds: 1), (_) async {
 103|       if (!mounted || !_isTracking) {
 104|         return;
 105|       }

 107|       if (lastUpdate != null &&
 108|           DateTime.now().difference(lastUpdate) <
 109|               const Duration(seconds: 3)) {
 110|         return;
 111|       }
 112|       await _fetchCurrentPosition();
 113|     });
 114|   }

 117|     _watchdog?.cancel();
 118|     _watchdog = null;
 119|   }

 121|   Future<void> _fetchCurrentPosition() async {
 122|     if (_isFetchingPosition || !_isTracking) {
 123|       return;
 124|     }
 125|     _isFetchingPosition = true;
 126|     try {
 127|       final position = await Geolocator.getCurrentPosition(
 128|         locationSettings: const LocationSettings(
 129|           accuracy: LocationAccuracy.bestForNavigation,
 130|           timeLimit: Duration(seconds: 3),
 131|         ),
 132|       );
 133|       if (mounted && _isTracking) {
 134|         _onPosition(position);
 135|       }
 136|     } catch (_) {
 137|       // 获取失败（如超时），由后续轮询自动重试
 138|     } finally {
 139|       _isFetchingPosition = false;
 140|     }
 141|   }

 143|   Future<void> _startTracking() async {
 144|     if (_isTracking) {
 145|       return;
 146|     }
 147| 
 148|     final serviceEnabled = await Geolocator.isLocationServiceEnabled();
 149|     if (!serviceEnabled) {
 150|       setState(() {
 151|         _statusMessage = '定位服务未开启，请在系统设置中打开';
 152|         _isTracking = false;
 153|       });
 154|       return;
 155|     }
 156| 
 157|     var permission = await Geolocator.checkPermission();
 158|     if (permission == LocationPermission.denied) {
 159|       permission = await Geolocator.requestPermission();
 160|     }
 161| 
 162|     if (permission == LocationPermission.denied ||
 163|         permission == LocationPermission.deniedForever) {
 164|       setState(() {
 165|         _statusMessage = '定位权限被拒绝，请在设置中允许访问位置';
 166|         _isTracking = false;
 167|       });
 168|       return;
 169|     }
 170| 
 171|     _positionSubscription?.cancel();
 172|     _positionSubscription =
 173|         Geolocator.getPositionStream(locationSettings: _locationSettings)
 174|             .listen(
 175|               _onPosition,
 176|               onError: (Object error) {
 177|                 _stopwatch.stop();
 178|                 _stopTicker();
 179|                 _stopWatchdog();
 180|                 _positionSubscription?.cancel();
 181|                 _positionSubscription = null;
 182|                 setState(() {
 183|                   _statusMessage = '定位数据流错误: $error';
 184|                   _isTracking = false;
 185|                   _speedMps = 0;
 186|                   _lastTimestamp = null;
 187|                   _lastPosition = null;
 188|                   _lastTrackedPosition = null;
 189|                 });
 190|               },
 191|             );
 192| 
 193|     _stopwatch.start();
 194|     _startTicker();
 195|     _startWatchdog();
 196| 
 197|     setState(() {
 198|       _statusMessage = '测速中...';
 199|       _isTracking = true;
 200|       _hasSession = true;
 201|     });
 202|   }

 243|   void _onPosition(Position position) {
 244|     _lastPositionUpdate = DateTime.now();
 245| 
 246|     if (_isTracking) {
 247|       final trackedPrev = _lastTrackedPosition;
 248|       if (trackedPrev != null) {
 249|         _distanceMeters += Geolocator.distanceBetween(
 250|           trackedPrev.latitude,
 251|           trackedPrev.longitude,
 252|           position.latitude,
 253|           position.longitude,
 254|         );
 255|       }
 256|       _lastTrackedPosition = position;
 257|     }
 258| 
 259|     double speedMps = position.speed;
 260| 
 261|     if (speedMps.isNaN || speedMps.isNegative || speedMps == 0) {
 262|       final prev = _lastPosition;
 263|       final prevTime = _lastTimestamp;
 264|       if (prev != null && prevTime != null) {
 265|         final now = DateTime.now();
 266|         final dt = now.difference(prevTime).inMilliseconds / 1000;
 267|         if (dt > 0) {
 268|           final distance = Geolocator.distanceBetween(
 269|             prev.latitude,
 270|             prev.longitude,
 271|             position.latitude,
 272|             position.longitude,
 273|           );
 274|           speedMps = distance / dt;
 275|         }
 276|         _lastTimestamp = now;
 277|       } else {
 278|         _lastTimestamp = DateTime.now();
 279|       }
 280|     } else {
 281|       _lastTimestamp = DateTime.now();
 282|     }
 283| 
 284|     _lastPosition = position;
 285| 
 286|     setState(() {
 287|       _speedMps = speedMps.clamp(0, 999);
 288|       _statusMessage = '测速中...';
 289|       _isTracking = true;
 290|     });
 291|   }

 604| class _SpeedGaugePainter extends CustomPainter {
 605|   _SpeedGaugePainter({
 606|     required this.speedKmh,
 607|     required this.maxSpeedKmh,
 608|     required this.colorScheme,
 609|   });
 610| 
 611|   final double speedKmh;
 612|   final double maxSpeedKmh;
 613|   final ColorScheme colorScheme;
 614| 
 615|   static const double _startAngle = math.pi * 0.75;
 616|   static const double _sweepAngle = math.pi * 1.5;
 617| 
 618|   // 非线性刻度：0-160 km/h 展开（占 72% 的弧长），160-400 km/h 压缩（占 28%）
 619|   static const double _breakpointKmh = 160;
 620|   static const double _breakpointFraction = 0.72;
 621| 
 622|   /// 把速度 km/h 映射到表盘的弧长比例 [0, 1]
 623|   double _speedToFraction(double kmh) {
 624|     final s = kmh.clamp(0.0, maxSpeedKmh);
 625|     if (s <= _breakpointKmh) {
 626|       return (s / _breakpointKmh) * _breakpointFraction;
 627|     }
 628|     return _breakpointFraction +
 629|         ((s - _breakpointKmh) / (maxSpeedKmh - _breakpointKmh)) *
 630|             (1 - _breakpointFraction);
 631|   }
 632| 
 633|   @override
 634|   void paint(Canvas canvas, Size size) {
 635|     final center = Offset(size.width / 2, size.height * 0.58);
 636|     final radius = math.min(size.width / 2 - 20, size.height * 0.46);
 637| 
 638|     final arcRect = Rect.fromCircle(center: center, radius: radius);
 639|     final value = _speedToFraction(speedKmh);
 640|     final needleAngle = _startAngle + _sweepAngle * value;
 641| 
 642|     final baseArcPaint = Paint()
 643|       ..color = colorScheme.surfaceContainerHighest
 644|       ..style = PaintingStyle.stroke
 645|       ..strokeWidth = 16
 646|       ..strokeCap = StrokeCap.round;
 647| 
 648|     final activeArcPaint = Paint()
 649|       ..shader = SweepGradient(
 650|         startAngle: _startAngle,
 651|         endAngle: _startAngle + _sweepAngle,
 652|         colors: const [Color(0xFF2E7D32), Color(0xFFF9A825), Color(0xFFC62828)],
 653|       ).createShader(arcRect)
 654|       ..style = PaintingStyle.stroke
 655|       ..strokeWidth = 16
 656|       ..strokeCap = StrokeCap.round;
 657| 
 658|     canvas.drawArc(arcRect, _startAngle, _sweepAngle, false, baseArcPaint);
 659|     canvas.drawArc(
 660|       arcRect,
 661|       _startAngle,
 662|       _sweepAngle * value,
 663|       false,
 664|       activeArcPaint,
 665|     );
 666| 
 667|     final majorTickPaint = Paint()
 668|       ..color = colorScheme.onSurface.withValues(alpha: 0.7)
 669|       ..strokeWidth = 2.4
 670|       ..strokeCap = StrokeCap.round;
 671| 
 672|     final majorTickActivePaint = Paint()
 673|       ..color = colorScheme.onSurface
 674|       ..strokeWidth = 2.4
 675|       ..strokeCap = StrokeCap.round;
 676| 
 677|     final minorTickPaint = Paint()
 678|       ..color = colorScheme.onSurface.withValues(alpha: 0.38)
 679|       ..strokeWidth = 1.4
 680|       ..strokeCap = StrokeCap.round;
 681| 
 682|     // 主刻度：0-160 每 20 km/h（展开、刻度更细），160-400 稀疏（200/250/300/350/400）
 683|     const List<double> majorTickSpeeds = [
 684|       0, 20, 40, 60, 80, 100, 120, 140, 160, 200, 250, 300, 350, 400,
 685|     ];
 686|     // 次刻度：仅在 0-160 内，每 10 km/h（主刻度之间各一条）
 687|     const List<double> minorTickSpeeds = [10, 30, 50, 70, 90, 110, 130, 150];
 688| 
 689|     for (final tickSpeed in majorTickSpeeds) {
